% House date style "D Month YYYY" is composed here, independent of biblatex language settings.
\makeatletter
\newcommand\waihona@urldate{\thefield{urlday}~\ifcase\thefield{urlmonth}\or\biblstring{january}\or\biblstring{february}\or\biblstring{march}\or\biblstring{april}\or\biblstring{may}\or\biblstring{june}\or\biblstring{july}\or\biblstring{august}\or\biblstring{september}\or\biblstring{october}\or\biblstring{november}\or\biblstring{december}\fi~\thefield{urlyear}}
\renewbibmacro*{url+urldate}{%
  \usebibmacro{url}%
  \iffieldundef{archiveurl}
    {\iffieldundef{urlyear}{}{\addspace(accessed~\waihona@urldate)}}
    {\addspace(%
       \iffieldundef{urlyear}{}{accessed~\waihona@urldate; }%
       \href{\thefield{archiveurl}}{archived~\thefield{archivedate}})}}
\makeatother
